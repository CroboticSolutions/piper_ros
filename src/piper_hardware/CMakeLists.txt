cmake_minimum_required(VERSION 3.16)
project(piper_hardware)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(piper_hardware SHARED src/piper_system.cpp)
target_compile_features(piper_hardware PUBLIC cxx_std_17)
ament_target_dependencies(piper_hardware hardware_interface pluginlib rclcpp rclcpp_lifecycle sensor_msgs std_msgs)
pluginlib_export_plugin_description_file(hardware_interface piper_hardware.xml)
install(TARGETS piper_hardware LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_piper_system test/test_piper_system.cpp)
  if(TARGET test_piper_system)
    target_link_libraries(test_piper_system piper_hardware)
    ament_target_dependencies(test_piper_system hardware_interface pluginlib rclcpp sensor_msgs std_msgs)
  endif()
endif()
ament_export_libraries(piper_hardware)
ament_export_dependencies(hardware_interface pluginlib rclcpp sensor_msgs std_msgs)
ament_package()
